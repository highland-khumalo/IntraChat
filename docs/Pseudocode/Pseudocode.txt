CLASS Client Handler

     DECLARE socket AS Socket
     DECLARE server AS Server
     DECLARE input Reader AS Buffered Reader
     DECLARE output Writer AS Print Writer
     DECLARE username AS String
     DECLARE password AS String

    METHOD Client Handler(socket, server)
        STORE socket
        STORE server
    TRY 
      CREATE input Reader using socket's input stream
      CREATE output Writer using socket's output stream
    END TRY
CATCH Commination error
      DISPLAY "Error while creating input/output streams"
    END CATCH
END METHOD

    METHOD run()
        TRY
        READ initial Name from input  Reader
        IF initial Name is not null
            SET username to initial Name
     END IF
        READ initial Password from input Reader
        IF initial Password is not null
            SET password to initial Password
        END IF
        BROADCAST that the user has joined the chat
        
        DECLARE incoming Message AS String
        WHILE incoming Message is not null
            READ incoming Message from input Reader
            IF incoming Message equals "/quit" THEN
             TERMINATE the communication loop
            ELSE 
            FORMAT message using username and incoming message
            RECORD the formatted message in the server log
            BROADCAST the formatted message to all connected clients
            END IF
        END WHILE
    CATCH Communication error
        DISPLAY a message indicating an error occurred while reading the message
    FINALLY
        CLOSE Connection
    END TRY 

   METHOD send Message(message)       
     IF output Writer is not null
        WRITE message to output Writer
    END IF
    END METHOD

     METHOD close Connection()
     REMOVE the client from the server's list of connected clients
     NOTIFY all connected clients that the user has left the chat
      TRY 
      IF input stream exists THEN
      CLOSE input stream
      END IF 
      IF output stream exists THEN
      CLOSE output stream
      END IF

      IF socket exists AND socket is not closed THEN
      CLOSE socket
      END IF
      CATCH Communication error
      DISPLAY error message indicating that the connection could not be closed properly
        END TRY
        END METHOD

        METHOD get Username()

        Access the client's username
        Provide the username to the calling method
    END METHOD
END CLASS

CLASS Server
    
    DECLARE port AS Integer
    DECLARE server Socket AS Server Socket
    DECLARE is Running AS Boolean
    DECLARE clients AS List of Client Handler
    
    METHOD Server(port)
        STORE port
        INITIALIZE clients as an empty list
    END METHOD
      
    METHOD start()
      SET is Running to TRUE
      TRY
         CREATE server Socket using the specified port
         DISPLAY "Server started on port " + port + " and waiting for clients..."
           WHILE is Running
           Wait for a client connection
           ACCEPT the client connection 
           CREATE a new Client Handler for the connected client
           ADD the new Client Handler to the list of connected clients
           START a new thread to handle communication with the client
        END WHILE

           CATCH Communication error 
           IF the server is still running THEN
            DISPLAY "Error while starting the server" 
        END IF
    FINALLY
          STOP the server and close all client connections
        END TRY
    END METHOD

    METHOD broadcast(message, sender)
        FOR EACH client IN clients
                SEND message to the client using send Message method
            END IF
        END FOR
    END METHOD

    METHOD remove Client(client)
    REMOVE the specified client from the list of connected clients
    DISPLAY the message indicating that the client has disconnected
    DISPLAY the total number of active clients
    END METHOD

    METHOD stop()
        SET is Running to FALSE
        TRY
            IF server Socket exists AND server Socket is not closed THEN
                CLOSE server Socket
            END IF
        CATCH Communication error
             FOR each connected client in clients
             CLOSE the client's connection
             END FOR
       
       CLEAR all the clients from the client list
            DISPLAY "Error while stopping the server"
        END TRY
    END METHOD
END CLASS
