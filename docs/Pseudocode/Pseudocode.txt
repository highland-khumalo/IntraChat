CLASS ClientHandler

     DECLARE socket AS Socket
     DECLARE server AS Server
     DECLARE inputReader AS BufferedReader
     DECLARE outputWriter AS PrintWriter
     DECLARE username AS String
     DECLARE password AS String

    METHOD ClientHandler(socket, server)
        STORE socket
        STORE server
    TRY 
      CREATE inputReader using socket's input stream
      CREATE outputWriter using socket's output stream
    END TRY
CATCH IOException
      DISPLAY "Error while creating input/output streams"
    END CATCH
END METHOD

    METHOD run()
        TRY
        READ initialName from inputReader
        IF initialName is not null
            SET username to initialName
     END IF
        READ initialPassword from inputReader
        IF initialPassword is not null
            SET password to initialPassword
        END IF
        BROADCAST that the user has joined the chat
        
        DECLARE incomingMessage AS String
        WHILE incomingMessage is not null
            READ incomingMessage from inputReader
            IF incomingMessage equals "/quit" THEN
             TERMINATE the communictaion loop
            ELSE 
            FORMAT message using username and incoming message
            RECORD the formatted message in the server log
            BROADCAST the formatted message to all connected clients
            END IF
        END WHILE
    CATCH IOException
        DISPLAY a message indicating an error occurred while reading the message
    FINALL
        CLOSE Connection
    END TRY 

   METHOD sendMessage(message)       
     IF outputWriter is not null
        WRITE message to outputWriter
    END IF
    END METHOD

     METHOD closeConnection()
     REMOVE the client from the server's list of connected clients
     NOTIFY all connected clients that the user has left the chat
      TRY 
      IF input stream exists THEN
      CLOSE input stream

      END IF

      IF socket exists AND socket is not closed THEN

      CATCH IOException

        END TRY
        END METHOD

CLASS Server
    
    DECLARE port AS Integer
    DECLARE serverSocket AS ServerSocket
    DECLARE isRunning AS Boolean
    DECLARE clients AS List of ClientHandler
    
    METHOD Server(port)
        STORE port
        INITIALIZE clients as an empty list
    END METHOD
      
    METHOD start()
      SET isRunning to TRUE
      TRY
         CREATE serverSocket using the specified port
         DISPLAY "Server started on port " + port + " and waiting for clients..."
           WHILE isRunning
           Wait for a client connection
           ACCEPT the client connection 
           CREATE a new ClientHandler for the connected client
           ADD the new ClientHandler to the list of connected clients
           START a new thread to handle communication with the client
        END WHILE

           CATCH IOException
           IF the serverr is still running THEN
            DISPLAY "Error while starting the server" 
        END IF
    FINALLY
          STOP the server and close all client connections
        END TRY
    END METHOD
